local animationsOn = true
local animSpeed = 1

local accentRefs = {}
local orders = {}
local W = {} -- widgets / fonctions partagées entre onglets

local function tw(obj, t, props, style, dir)
	local d = animationsOn and (t / animSpeed) or 0
	local info = TweenInfo.new(d, style or ES.Quad, dir or ED.Out)
	local tween = TweenService:Create(obj, info, props)
	tween:Play()
	return tween
end

local function new(class, props, parent)
	local o = Instance.new(class)
	for k, v in pairs(props) do
		o[k] = v
	end
	if parent then
		o.Parent = parent
	end
	return o
end

local function corner(obj, r)
	return new("UICorner", { CornerRadius = UDim.new(0, r or 8) }, obj)
end

local function round(obj)
	return new("UICorner", { CornerRadius = UDim.new(0.5, 0) }, obj)
end

local function ord(p)
	orders[p] = (orders[p] or 0) + 1
	return orders[p]
end

local function accent(obj, prop)
	table.insert(accentRefs, { obj, prop })
	obj[prop] = accentColor
	return obj
end

local function setAccent(c)
	accentColor = c
	for _, r in ipairs(accentRefs) do
		if r[1] and r[1].Parent then
			r[1][r[2]] = c
		end
	end
	if W.refreshTabs then W.refreshTabs() end
	if W.refreshShaders then W.refreshShaders() end
end

local function hover(btn, blocked)
	btn.AutoButtonColor = false
	btn.MouseEnter:Connect(function()
		if blocked and blocked() then return end
		tw(btn, 0.12, { BackgroundTransparency = 0.15 })
	end)
	btn.MouseLeave:Connect(function()
		if blocked and blocked() then return end
		tw(btn, 0.12, { BackgroundTransparency = 0 })
	end)
	btn.MouseButton1Down:Connect(function()
		if blocked and blocked() then return end
		tw(btn, 0.05, { BackgroundTransparency = 0.4 })
	end)
	btn.MouseButton1Up:Connect(function()
		if blocked and blocked() then return end
		tw(btn, 0.1, { BackgroundTransparency = 0.15 })
	end)
end

local currentStretchY = 1
local stretchConnection = nil -- conservé pour compatibilité avec le reste du script
local function applyResolution(ratioY)
	-- Roblox ne permet pas à un LocalScript de changer le vrai ratio de rendu.
	-- Éviter de multiplier Camera.CFrame par une matrice d'échelle à chaque frame :
	-- cela déforme la transformation de la caméra au lieu de changer la résolution.
	currentStretchY = math.clamp(tonumber(ratioY) or 1, 0.1, 1)
	if stretchConnection then
		stretchConnection:Disconnect()
		stretchConnection = nil
	end
end

RunService.RenderStepped:Connect(function(dt)
	if fpsLabel.Visible then
		fpsFrames += 1
		fpsAcc += dt
		if fpsAcc >= 0.5 then
			local fps = math.floor(fpsFrames / fpsAcc + 0.5)
			local ping = 0
			pcall(function()
				ping = math.floor(player:GetNetworkPing() * 1000 + 0.5)
			end)
			fpsLabel.Text = "FPS: " .. fps .. "  |  Ping: " .. ping .. " ms"
			fpsFrames, fpsAcc = 0, 0
		end
	end

	if C.enabled and not menuOpen then
		local desired
		if C.mode == "Mouse" or C.mode == "Both" then
			desired = false
		else
			desired = panelOpen
		end
		if UserInputService.MouseIconEnabled ~= desired then
			UserInputService.MouseIconEnabled = desired
		end
	end

	local showCenter = C.enabled and not menuOpen and (C.mode == "Center" or C.mode == "Both")
	local showMouse = C.enabled and not menuOpen and (C.mode == "Mouse" or C.mode == "Both")

	local color = C.rgb and Color3.fromHSV((os.clock() % 4) / 4, 1, 1) or C.color
	local boxSize = math.max(50, C.size * 3)
	local transp = 1 - C.opacity / 100
	local strokeT = C.outline and transp or 1

	local function style(label)
		label.Text = shapesList[C.shape]
		label.TextSize = C.size
		label.Size = UDim2.new(0, boxSize, 0, boxSize)
		label.TextColor3 = color
		label.TextTransparency = transp
		label.TextStrokeTransparency = strokeT
	end

	centerCross.Visible = showCenter
	if showCenter then
		style(centerCross)
		local cam = workspace.CurrentCamera
		if cam then
			local vs = cam.ViewportSize
			centerCross.Position = UDim2.new(0, vs.X / 2, 0, vs.Y / 2)
		end
	end

	mouseCross.Visible = showMouse
	if showMouse then
		style(mouseCross)
		local loc = UserInputService:GetMouseLocation()
		mouseCross.Position = UDim2.new(0, loc.X, 0, loc.Y)
	end
end)

local function toast(msg)
	if not notifOn then return end
	toastToken += 1
	local my = toastToken
	toastLabel.Text = msg
	toastFrame.Visible = true
	toastFrame.Position = UDim2.new(0.5, 0, 1, 60)
	tw(toastFrame, 0.35, { Position = UDim2.new(0.5, 0, 1, -30) }, ES.Back, ED.Out)
	task.delay(2.4, function()
		if my ~= toastToken then return end
		local t = tw(toastFrame, 0.3, { Position = UDim2.new(0.5, 0, 1, 60) }, ES.Quad, ED.In)
		t.Completed:Wait()
		if my == toastToken then
			toastFrame.Visible = false
		end
	end)
end

local function makeDraggable(handle, frame)
	local dragging, dragStart, startPos = false, nil, nil
	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = frame.Position
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local d = input.Position - dragStart
			frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
end

local function makeButton(page, text, cb, kind, h)
	local b = new("TextButton", {
		Size = UDim2.new(1, 0, 0, h or 34),
		BackgroundColor3 = LIGHT,
		Text = text,
		TextColor3 = TXT,
		TextSize = 12,
		Font = EF.GothamBold,
		LayoutOrder = ord(page),
	}, page)
	if kind == "accent" then
		accent(b, "BackgroundColor3")
		b.TextColor3 = WHITE
	elseif kind == "red" then
		b.BackgroundColor3 = RED
		b.TextColor3 = WHITE
	end
	corner(b, 8)
	hover(b)
	if cb then
		b.MouseButton1Click:Connect(function()
			cb(b)
		end)
	end
	return b
end

		corner(b, 6)
		hover(b)
		b.MouseButton1Click:Connect(function()
			d[2](b, i)
		end)
		btns[i] = b
	end
	return btns
end

local function makeToggle(page, text, default, cb)
	local state = false
	local b = makeButton(page, "", nil, nil, 34)
	local obj = { button = b }
	local function refresh()
		b.BackgroundColor3 = state and GREEN or RED
		b.TextColor3 = WHITE
		b.Text = text .. " : " .. (state and "ON" or "OFF")
	end
	function obj.set(v, silent)
		state = v and true or false
		refresh()
		if not silent and cb then
			cb(state)
		end
	end
	function obj.get()
		return state
	end
	b.MouseButton1Click:Connect(function()
		obj.set(not state)
	end)
	obj.set(default, true)
	return obj
end

local activeSlider = nil
local function makeSlider(page, text, min, max, default, step, cb, fmt)
	local holder = new("Frame", {
		Size = UDim2.new(1, 0, 0, 46),
		BackgroundTransparency = 1,
		LayoutOrder = ord(page),
	}, page)
	local lbl = new("TextLabel", {
		Size = UDim2.new(1, 0, 0, 18),
		BackgroundTransparency = 1,
		TextColor3 = SUB,
		TextSize = 12,
		Font = EF.GothamMedium,
		TextXAlignment = Enum.TextXAlignment.Left,
	}, holder)
	local bg = new("Frame", {
		Size = UDim2.new(1, -10, 0, 10),
		Position = UDim2.new(0, 5, 0, 30),
		BackgroundColor3 = TRACK,
		BorderSizePixel = 0,
	}, holder)
	corner(bg, 5)
	local fill = new("Frame", { Size = UDim2.new(0, 0, 1, 0), BorderSizePixel = 0 }, bg)
	accent(fill, "BackgroundColor3")
	corner(fill, 5)
	local knob = new("Frame", {
		Size = UDim2.new(0, 18, 0, 18),
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.new(0, 0, 0.5, 0),
		BackgroundColor3 = WHITE,
	}, bg)
	corner(knob, 9)
	local ks = new("UIStroke", { Thickness = 2 }, knob)
	accent(ks, "Color")
	local hit = new("TextButton", {
		Size = UDim2.new(1, 0, 0, 28),
		Position = UDim2.new(0, 0, 0, 18),
		BackgroundTransparency = 1,
		Text = "",
		ZIndex = 5,
	}, holder)

	local value = default
	local obj = {}
	local function render()
		local a = (value - min) / (max - min)
		fill.Size = UDim2.new(a, 0, 1, 0)
		knob.Position = UDim2.new(a, 0, 0.5, 0)
		lbl.Text = text .. " : " .. (fmt and fmt(value) or tostring(value))
	end
	function obj.set(v, silent)
		v = math.clamp(v, min, max)
		v = math.floor(v / step + 0.5) * step
		value = tonumber(string.format("%.4f", v))
		render()
		if not silent and cb then
			cb(value)
		end
	end
	function obj.get()
		return value
	end
	function obj.fromX(x)
		local a = math.clamp((x - bg.AbsolutePosition.X) / math.max(bg.AbsoluteSize.X, 1), 0, 1)
		obj.set(min + a * (max - min))
	end
	hit.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			activeSlider = obj
			obj.fromX(input.Position.X)
		end
	end)
	obj.set(default, true)
	return obj
end

UserInputService.InputChanged:Connect(function(input)
	if activeSlider and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		activeSlider.fromX(input.Position.X)
	end
end)
UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		activeSlider = nil
	end
end)

local function makeColorGrid(page, colors, onPick)
	local grid = new("Frame", {
		Size = UDim2.new(1, 0, 0, 30), BackgroundTransparency = 1, LayoutOrder = ord(page),
	}, page)
	new("UIGridLayout", {
		CellSize = UDim2.new(0, 36, 0, 26), CellPadding = UDim2.new(0, 6, 0, 4),
		SortOrder = Enum.SortOrder.LayoutOrder,
	}, grid)
	for idx, col in ipairs(colors) do
		local b = new("TextButton", { BackgroundColor3 = col, Text = "", LayoutOrder = idx }, grid)
		corner(b, 6)
		hover(b)
		b.MouseButton1Click:Connect(function()
			onPick(col)
		end)
	end
	-- hauteur suffisante pour plusieurs lignes
	local perRow = 8
	local rows = math.ceil(#colors / perRow)
	grid.Size = UDim2.new(1, 0, 0, rows * 30)
	return grid

	for i, l in ipairs(stepLabels) do
		if p >= stepThresholds[i] then
			l.Text = "✓ " .. stepNames[i]
			l.TextColor3 = GREEN
		elseif not currentFound then
			currentFound = true
			l.Text = "● " .. stepNames[i]
			l.TextColor3 = WHITE
		else
			l.Text = "○ " .. stepNames[i]
			l.TextColor3 = Color3.fromRGB(110, 120, 145)
		end
	end

do
	local p = pages[2]
	makeHeading(p, "🥔 Potato")

	local potatoEnabled = false
	local potatoLevel = 100
	local originalState = {}
	local potatoToken = 0

	local function applyPotatoGraphics()
		if not potatoEnabled then return end
		potatoToken += 1
		local my = potatoToken
		local factor = potatoLevel / 100
		local count = 0
		for _, obj in ipairs(workspace:GetDescendants()) do
			if my ~= potatoToken or not potatoEnabled then return end
			local model = obj:FindFirstAncestorOfClass("Model")
			if not model or not Players:GetPlayerFromCharacter(model) then
				if obj:IsA("BasePart") then
					if not originalState[obj] then
						originalState[obj] = { material = obj.Material }
					end
					if factor >= 0.8 then
						obj.Material = Enum.Material.SmoothPlastic
					elseif factor >= 0.4 then
						obj.Material = Enum.Material.Plastic
					else
						obj.Material = originalState[obj].material
					end
				elseif obj:IsA("Decal") or obj:IsA("Texture") then
					if not originalState[obj] then
						originalState[obj] = { transparency = obj.Transparency }
					end
					local t0 = originalState[obj].transparency
					obj.Transparency = math.clamp(t0 + (1 - t0) * factor, 0, 1)
				end
			end
			count += 1
			if count % 300 == 0 then
				task.wait()
			end
		end
	end

	local function restoreOriginalGraphics()
		potatoToken += 1
		for obj, state in pairs(originalState) do
			if obj and obj.Parent then
				if state.material and obj:IsA("BasePart") then
					obj.Material = state.material
				end
				if state.transparency and (obj:IsA("Decal") or obj:IsA("Texture")) then
					obj.Transparency = state.transparency
				end
			end
		end
		originalState = {}
	end

	W.potatoToggle = makeToggle(p, "🥔 Mode Potato", false, function(on)
		potatoEnabled = on
		if on then
			task.spawn(applyPotatoGraphics)
		else
			restoreOriginalGraphics()
		end
	end)

	local pending = false
	W.potatoSlider = makeSlider(p, "Niveau Potato (réduction de texture)", 0, 100, 100, 1, function(v)
		potatoLevel = v
		if potatoEnabled and not pending then

	makeLabel(p, "Choisis un shader (un seul à la fois, clique à nouveau pour l'éteindre) :", 12, false, SUB, 34)

	local function mk(class, name, props)
		props.Name = "PanelModShader_" .. name
		props.Enabled = false
		local old = lighting:FindFirstChild(props.Name)
		if old then old:Destroy() end
		return new(class, props, lighting)
	end
	local sCC = mk("ColorCorrectionEffect", "CC", {})
	local sBloom = mk("BloomEffect", "Bloom", { Intensity = 0, Size = 24, Threshold = 2 })
	local sBlur = mk("BlurEffect", "Blur", { Size = 0 })
	local sRays = mk("SunRaysEffect", "Rays", { Intensity = 0, Spread = 1 })
	local sDof = mk("DepthOfFieldEffect", "DoF", {
		FarIntensity = 0, NearIntensity = 0, FocusDistance = 50, InFocusRadius = 50,
	})

	local function lerp(a, b, t) return a + (b - a) * t end

	local function neutral()
		sCC.Enabled = false

	local function refreshShaders()
		for i, s in ipairs(SHADERS) do
			local sel = (s.id == activeId)
			shaderBtns[i].BackgroundColor3 = sel and accentColor or LIGHT
			shaderBtns[i].TextColor3 = sel and WHITE or TXT
		end

			descLabel.Text = activeId and byId[activeId].desc or "Aucun shader actif."
		end
	end

		neutral()

		blurSize = 0

			byId[activeId].setup(intensity / 100)

		refreshShaders()

		elseif activeId == "Reve" and animateOn then
			sBloom.Intensity = lerp(0, 1.5, k) + math.sin(os.clock() * 1.5) * 0.25 * k
		elseif activeId == "Mouvement" then

			if cam and dt > 0 then
				local look = cam.CFrame.LookVector
				local pos = cam.CFrame.Position
				local target = 0
				if lastLook then
					local ang = math.acos(math.clamp(look:Dot(lastLook), -1, 1))
					local angSpeed = ang / dt
					local linSpeed = (pos - lastPos).Magnitude / dt
					target = math.clamp(angSpeed * 2.5 + linSpeed * 0.05, 0, 24) * k
				end
				lastLook, lastPos = look, pos
				blurSize = blurSize + (target - blurSize) * math.clamp(dt * 10, 0, 1)
				sBlur.Size = blurSize
			end
		end
	end)

	function W.resetShaders()

		W.shaderAnim.set(true, true)
		animateOn = true
		applyShader()
	end

-- Raccourci clavier du panel et capture de la nouvelle touche.
UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if listeningForKey then
		if input.UserInputType ~= Enum.UserInputType.Keyboard then return end
		if input.KeyCode == Enum.KeyCode.Unknown then return end

		if input.KeyCode == Enum.KeyCode.Escape then
			listeningForKey = false
			W.keybindButton.Text = "Touche : " .. currentToggleKey.Name
			W.keybindButton.BackgroundColor3 = accentColor
			return
		end

		currentToggleKey = input.KeyCode
		listeningForKey = false
		W.keybindButton.Text = "Touche : " .. currentToggleKey.Name
		W.keybindButton.BackgroundColor3 = accentColor
		toast("Raccourci modifié : " .. currentToggleKey.Name)
		return
	end

	if gameProcessed or menuOpen then return end
	if UserInputService:GetFocusedTextBox() then return end
	if input.UserInputType ~= Enum.UserInputType.Keyboard then return end
	if input.KeyCode ~= currentToggleKey then return end

	if not panelUnlocked then
		toast("Clique d'abord sur la flèche pour initialiser le panel")
		return
	end

	if panelOpen then
		closePanel()
	else
		openPanel()
	end
end)

-- Animation d'initialisation locale : elle ne télécharge aucun fichier externe.
local function createConfetti()
	local confettiColors = {
		Color3.fromRGB(53, 120, 229), Color3.fromRGB(46, 204, 113),
		Color3.fromRGB(255, 220, 70), Color3.fromRGB(230, 126, 34),
		Color3.fromRGB(200, 80, 255),
	}
	for i = 1, 22 do
		local piece = new("Frame", {
			Name = "ConfettiPiece",
			Size = UDim2.new(0, math.random(4, 8), 0, math.random(6, 12)),
			Position = UDim2.new(math.random(5, 95) / 100, 0, 0, -10),
			BackgroundColor3 = confettiColors[math.random(1, #confettiColors)],
			BorderSizePixel = 0,
			Rotation = math.random(0, 180),
			ZIndex = 20,
		}, updateWindow)
		tw(piece, math.random(8, 16) / 10, {
			Position = UDim2.new(piece.Position.X.Scale, math.random(-25, 25), 0, 340),
			Rotation = math.random(-360, 360),
			BackgroundTransparency = 1,
		}, ES.Quad, ED.In)
		task.delay(1.8, function()
			if piece and piece.Parent then piece:Destroy() end
		end)
	end
end

	progressBar.Size = UDim2.new(0, 0, 1, 0)

			progressBar.Size = UDim2.new(progress, 0, 1, 0)
